## Using Web Cookies for Mobile API

```python
import time, ssl, urllib.request, urllib.parse, gzip, json
from ttk.signing_engine import sign

def send_view_with_web_cookies(video_id, web_cookies, device_info):
    ts = int(time.time())
    
    # Build params
    params = urllib.parse.urlencode({
        'play_delta': '1', 'item_id': video_id, 'aweme_type': '0',
        'os_api': '34', 'device_type': '2311DRK48C', 'ssmix': 'a',
        # ... all device params ...
    })
    
    body = f'play_delta=1&item_id={video_id}&aweme_type=0'.encode()
    host = 'api16-normal-c-alisg.tiktokv.com'
    full_url = f'https://{host}/aweme/v1/aweme/stats?{params}'
    sig = sign(full_url, method='POST', body=body, ts=ts)
    
    # Build cookie with web cookies
    cookie = (
        f'ttwid={web_cookies["ttwid"]}; '
        f'msToken={web_cookies["msToken"]}; '
        f'store-idc=useast5; store-country-code=vn; '
        f'install_id={device_info["install_id"]}'
    )
    
    # Build headers
    h = {
        'host': host,
        'user-agent': 'com.zhiliaoapp.musically/440315 (Linux; U; Android 14; vi_VN)',
        'x-ss-req-ticket': str(int(time.time() * 1000)),
        'content-type': 'application/x-www-form-urlencoded; charset=UTF-8',
        'cookie': cookie,
    }
    for k in ('X-Gorgon', 'X-Khronos', 'X-Ladon', 'X-Argus', 'X-SS-STUB'):
        if k in sig:
            h[k] = sig[k]
    
    # Send request
    ctx = ssl.create_default_context()
    ctx.check_hostname = False
    ctx.verify_mode = ssl.CERT_NONE
    opener = urllib.request.build_opener(urllib.request.HTTPSHandler(context=ctx))
    
    req = urllib.request.Request(full_url, data=body, headers=h, method='POST')
    resp = opener.open(req, timeout=15)
    raw = resp.read()
    if resp.headers.get('Content-Encoding') == 'gzip':
        raw = gzip.decompress(raw)
    
    if raw:
        j = json.loads(raw)
        return j.get('status_code') == 0, j
    return True, {'_empty': True}
```

## Key Insights

1. **Web cookies bypass odin_tt** — TikTok accepts `ttwid` + `msToken` instead
2. **Refresh cookies every 10 requests** — prevents staleness
3. **Same device_id = dedup** — even with different cookies, 1 device = 1 view max
4. **No proxy needed** — web cookies work from any IP
